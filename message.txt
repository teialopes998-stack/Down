local function createScriptCard(title, description, icon, keybindName, keybindTable, keybindKey)
    local Card = Instance.new("Frame")

    CardBorder.Parent = Card
    
    -- Ícone do script (menor)
    local CardIcon = Instance.new("TextLabel")
    CardIcon.Size = UDim2.new(0, 40, 0, 40) -- Tamanho reduzido
    CardIcon.Position = UDim2.new(0, 10, 0, 10) -- Posição ajustada
    CardIcon.BackgroundTransparency = 1
    CardIcon.Text = icon
    CardIcon.TextColor3 = themeColors.secondary
    CardIcon.TextSize = 22 -- Texto menor

    CardIcon.Parent = Card
    
    -- Título do script (menor)
    local CardTitle = Instance.new("TextLabel")
    CardTitle.Size = UDim2.new(0.6, 0, 0, 25) -- Tamanho reduzido

    CardTitle.BackgroundTransparency = 1
    CardTitle.Text = title
    CardTitle.TextColor3 = themeColors.light
    CardTitle.TextSize = 16 -- Texto menor
    CardTitle.Font = Enum.Font.GothamBold
    CardTitle.TextXAlignment = Enum.TextXAlignment.Left
    CardTitle.Parent = Card
    
    -- Descrição do script (menor)
    local CardDesc = Instance.new("TextLabel")
    CardDesc.Size = UDim2.new(0.8, 0, 0, 30) -- Tamanho reduzido
    CardDesc.Position = UDim2.new(0, 60, 0, 35) -- Posição ajustada
    CardDesc.BackgroundTransparency = 1
    CardDesc.Text = description
    CardDesc.TextColor3 = Color3.fromRGB(200, 200, 200)

    CardDesc.Font = Enum.Font.Gotham

    CardDesc.Parent = Card

    -- Função para atualizar o display do keybind
    local function updateKeybindDisplay()
        KeybindDisplay.Text = keybindTable[keybindKey].Name
    end
    
    -- Inicializar display
    updateKeybindDisplay()

    KeybindDisplay.MouseButton1Click:Connect(function()
        selectingKeybind = true
        KeybindDisplay.Text = "..."
        KeybindDisplay.BackgroundColor3 = themeColors.accent

        connection = UserInputService.InputBegan:Connect(function(input)
            if selectingKeybind and (input.KeyCode ~= Enum.KeyCode.Unknown) then
                selectingKeybind = false
                keybindTable[keybindKey] = input.KeyCode
                updateKeybindDisplay()
                KeybindDisplay.BackgroundColor3 = themeColors.primary
                if connection then
                    connection:Disconnect()
                end
            end
        end)
        
        -- Cancelar seleção após 5 segundos
        task.delay(5, function()
            if selectingKeybind then
                selectingKeybind = false
                updateKeybindDisplay()
                KeybindDisplay.BackgroundColor3 = themeColors.primary
                if connection then
                    connection:Disconnect()
                end
            end
        end)
    end)

    KeybindDisplay.MouseEnter:Connect(function()
        if not selectingKeybind then
            local tween = TweenService:Create(KeybindDisplay, TweenInfo.new(0.2), {
                BackgroundColor3 = themeColors.secondary
            })
            tween:Play()
        end
    end)

        if not selectingKeybind then
            local tween = TweenService:Create(KeybindDisplay, TweenInfo.new(0.2), {
                BackgroundColor3 = themeColors.primary
            })
            tween:Play()
        end
    end)
    
    -- Adicionar efeito de hover ao botão ATIVAR
    local originalColor = CardButton.BackgroundColor3
    CardButton.MouseEnter:Connect(function()
        local tween = TweenService:Create(CardButton, TweenInfo.new(0.2), {
            BackgroundColor3 = themeColors.accent
        })
        tween:Play()
    end)
    
    CardButton.MouseLeave:Connect(function()
        local tween = TweenService:Create(CardButton, TweenInfo.new(0.2), {
            BackgroundColor3 = originalColor
        })
        tween:Play()
    end)
    
    return Card, CardButton
end

local speedEnabled = false
local speedMultiplier = 1.8
local originalWalkSpeed = 16
local originalJumpPower = 50

local SpeedCard, SpeedButton = createScriptCard(
    "SPEED BOOST",
    "Real SPEED BOOST",
    "⚡",
    "Q",
    keybinds,
    "speed"
)

SpeedCard.Parent = ScriptsFrame
SpeedCard.LayoutOrder = 1

local function toggleSpeed()
    speedEnabled = not speedEnabled
    
    if not player.Character then return end
    local humanoid = player.Character:FindFirstChild("Humanoid")
    if not humanoid then return end
    
    if originalWalkSpeed == 16 then
        originalWalkSpeed = humanoid.WalkSpeed
        originalJumpPower = humanoid.JumpPower
    end
    
    if speedEnabled then
        humanoid.WalkSpeed = originalWalkSpeed * speedMultiplier
        humanoid.JumpPower = originalJumpPower * 1.2
        SpeedButton.Text = "ON"
        SpeedButton.BackgroundColor3 = themeColors.accent
    else
        humanoid.WalkSpeed = originalWalkSpeed
        humanoid.JumpPower = originalJumpPower
        SpeedButton.Text = "ATIVAR"
        SpeedButton.BackgroundColor3 = themeColors.primary
    end
end

SpeedButton.MouseButton1Click:Connect(toggleSpeed)

-- Função do helicóptero
local function updateHelicopter()
    if not player.Character then return end
    local HRP = player.Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end
    
    if heliEnabled then
        if not heliAttachment then
            heliAttachment = Instance.new("Attachment")
            heliAttachment.Parent = HRP
            
            heliAngularVelocity = Instance.new("AngularVelocity")
            heliAngularVelocity.Attachment0 = heliAttachment
            heliAngularVelocity.RelativeTo = Enum.ActuatorRelativeTo.Attachment0
            heliAngularVelocity.MaxTorque = math.huge
            heliAngularVelocity.Parent = HRP
        end
        
        heliAngularVelocity.AngularVelocity = Vector3.new(0, heliSpeeds[heliLevel], 0)
        heliAngularVelocity.Enabled = true
    else
        if heliAngularVelocity then
            heliAngularVelocity.Enabled = false
            heliAngularVelocity.AngularVelocity = Vector3.zero
            HRP.AssemblyAngularVelocity = Vector3.zero
            HRP.AssemblyLinearVelocity = Vector3.zero
        end
    end
end

local function toggleHelicopter()
    heliEnabled = not heliEnabled
    HeliButton.Text = heliEnabled and "ON" or "ATIVAR"
    HeliButton.BackgroundColor3 = heliEnabled and themeColors.accent or themeColors.primary
    updateHelicopter()
end

HeliButton.MouseButton1Click:Connect(toggleHelicopter)

HeliLevelButton.MouseButton1Click:Connect(function()
    heliLevel = heliLevel + 1
    if heliLevel > #heliSpeeds then
        heliLevel = 1
    end
    HeliLevelButton.Text = "NÍVEL " .. heliLevel
    if heliEnabled then
        updateHelicopter()
    end
end)

-- Função do auto bastão
local function getBastao()
    local character = player.Character
    if not character then return nil end
    for _, tool in ipairs(character:GetChildren()) do
        if tool:IsA("Tool") then
            return tool
        end
    end
    return nil
end

task.spawn(function()
    while true do
        if bastaoEnabled then
            local bastao = getBastao()
            if bastao then
                pcall(function()
                    bastao:Activate()
                end)
            end
        end
        task.wait(bastaoInterval)
    end
end)

local function toggleBastao()
    bastaoEnabled = not bastaoEnabled
    BastaoButton.Text = bastaoEnabled and "ON" or "ATIVAR"
    BastaoButton.BackgroundColor3 = bastaoEnabled and themeColors.accent or themeColors.primary
end

    local animate = character:FindFirstChild("Animate")
    if animate and animate:IsA("LocalScript") then
        originalAnimateScript = animate:Clone()
    end
    
    -- Remover script Animate
    if animate then
        animate:Destroy()
    end
    
    -- Remover outras animações
    local humanoid = character:FindFirstChild("Humanoid")
    if humanoid then
        -- Parar todas as animações ativas
        for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
            track:Stop()
        end
    end
    
    -- Desabilitar scripts de animação adicionais
    for _, child in ipairs(character:GetDescendants()) do
        if child:IsA("LocalScript") and (child.Name:find("Anim") or child.Name:find("Animation")) then
            child:Destroy()
        end
    end
end

-- Função para restaurar animação
local function restaurarAnimacao()
    if not player.Character then return end
    
    local character = player.Character
    
    -- Restaurar script Animate se tivermos salvo
    if originalAnimateScript then
        local newAnimate = originalAnimateScript:Clone()
        newAnimate.Parent = character
        newAnimate.Disabled = false
    end
end

-- Função para alternar animação
local function toggleNoAnim()
    noAnimEnabled = not noAnimEnabled
    
    if noAnimEnabled then
        -- Remover animação
        removerAnimacao()
        NoAnimButton.Text = "ON"
        NoAnimButton.BackgroundColor3 = themeColors.red
    else
        -- Restaurar animação
        restaurarAnimacao()
        NoAnimButton.Text = "ATIVAR"
        NoAnimButton.BackgroundColor3 = themeColors.primary
    end
end
